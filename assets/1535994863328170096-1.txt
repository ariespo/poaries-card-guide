  <script>
  (function () {
    const wrapper =
      (document.currentScript && document.currentScript.closest('.th-status-wrapper')) ||
      Array.from(document.querySelectorAll('.th-status-wrapper:not([data-th-ready])')).pop();
    if (!wrapper) {
      console.error('[状态栏] 未找到面板容器，请确认 <script> 位于 .th-status-wrapper 内部');
      return;
    }
    wrapper.setAttribute('data-th-ready', '1');

    const $ = name => wrapper.querySelector('[data-th="' + name + '"]');

    // 脱离酒馆环境（直接双击打开 .html）时使用的演示数据
    const DEMO = {
      好感度: 132,
      关系阶段: '暧昧',
      心情: { 状态: '雀跃', 内心所想: '今天他的伞倾向了我这边三厘米……我数过了。' },
      穿着: '白色衬衫、深色外套、深色百褶裙、黑色长筒袜、小皮鞋',
      爱之日记: {
        '六月十一日': '佐藤同学又恬不知耻地凑了上去呢。\n那个位置是我的。不要用你那肮脏的手触碰他。',
        '六月十二日': '没关系的。从今以后你不需要再出现了。\n这里只有我们两个。我爱你。',
      },
      阶段: 2,
    };

    const unwrap = v => (Array.isArray(v) ? v[0] : v);
    const dig = (obj, path, fallback) => {
      let cur = obj;
      for (const key of path.split('.')) {
        if (cur === null || typeof cur !== 'object' || !(key in cur)) return fallback;
        cur = unwrap(cur[key]);
      }
      return cur === undefined || cur === null ? fallback : cur;
    };

    function readStat() {
      // 浏览器里没有酒馆助手接口，回退到演示数据
      if (typeof getVariables !== 'function') return DEMO;

      let vars = {};
      try {
        vars = getVariables({ type: 'message', message_id: getCurrentMessageId() }) || {};
      } catch (e) {
        try { vars = getVariables({ type: 'chat' }) || {}; } catch (e2) { vars = {}; }
      }
      const root = dig(vars, 'stat_data.时雨天音', null);
      return root && typeof root === 'object' ? root : {};
    }

    const tierOf = v =>
      v >= 200 ? '狂热' : v >= 150 ? '爱恋' : v >= 100 ? '喜欢' :
      v >= 50 ? '颇有好感' : v >= 0 ? '陌生' : v >= -50 ? '厌恶' : '极度厌恶';

    function render() {
      const stat = readStat();

      const favor = Math.max(-100, Math.min(200, Number(unwrap(stat['好感度'])) || 0));
      $('好感度数值').textContent = favor;
      $('好感度称谓').textContent = tierOf(favor);
      $('好感度条').style.width = ((favor + 100) / 300) * 100 + '%';

      $('关系阶段').textContent = String(dig(stat, '关系阶段', '暗恋'));
      $('心情状态').textContent = String(dig(stat, '心情.状态', '平静'));
      $('内心所想').textContent = String(dig(stat, '心情.内心所想', '……'));

      const tags = $('穿着');
      tags.textContent = '';
      String(dig(stat, '穿着', ''))
        .split(/[、,，;；]/)
        .map(s => s.trim())
        .filter(Boolean)
        .forEach(item => {
          const span = document.createElement('span');
          span.className = 'thsw-tag';
          span.textContent = item;
          tags.appendChild(span);
        });

      const unlocked = Number(unwrap(stat['阶段'])) === 2;
      $('日记入口').hidden = !unlocked;
      if (!unlocked) {
        $('日记面板').hidden = true;
        return;
      }

      const body = $('日记内容');
      body.textContent = '';
      const diary = dig(stat, '爱之日记', {}) || {};
      const entries = Object.entries(diary).reverse();
      if (!entries.length) {
        const p = document.createElement('div');
        p.className = 'thsw-empty';
        p.textContent = '这里还什么都没有写下……';
        body.appendChild(p);
        return;
      }
      entries.forEach(([date, text]) => {
        const item = document.createElement('div');
        item.className = 'thsw-entry';
        const d = document.createElement('div');
        d.className = 'thsw-entry-date';
        d.textContent = date;
        const t = document.createElement('div');
        t.className = 'thsw-entry-text';
        t.textContent = String(unwrap(text));
        item.appendChild(d);
        item.appendChild(t);
        body.appendChild(item);
      });
    }

    wrapper.querySelector('.thsw-diary-btn').addEventListener('click', () => {
      const panel = $('日记面板');
      panel.hidden = !panel.hidden;
    });
    wrapper.querySelector('.thsw-diary-close').addEventListener('click', () => {
      $('日记面板').hidden = true;
    });

    try {
      render();
    } catch (e) {
      const box = document.createElement('pre');
      box.style.cssText = 'padding:8px;color:#ff9ec0;font-size:11px;white-space:pre-wrap;';
      box.textContent = '状态栏渲染出错：' + e.message;
      wrapper.appendChild(box);
    }

    if (typeof eventOn === 'function') {
      try {
        eventOn('mag_variable_update_ended', render);
        eventOn(tavern_events.MESSAGE_UPDATED, render);
        eventOn(tavern_events.MESSAGE_SWIPED, render);
      } catch (e) {}
    }
  })();
  </script>
