<script>
    let lastRawText = '';
    let isOptionsExpanded = false;

    // 切换选项框展开/收缩状态
    function toggleOptions() {
        isOptionsExpanded = !isOptionsExpanded;
        const optionsList = document.getElementById('optionsList');
        const icon = document.getElementById('optionsToggleIcon');
        
        if (isOptionsExpanded) {
            optionsList.style.display = 'flex';
            icon.className = 'fa-solid fa-chevron-up';
        } else {
            optionsList.style.display = 'none';
            icon.className = 'fa-solid fa-chevron-down';
        }
    }

    // 核心显示更新逻辑
    async function updateDisplay() {
        try {
            // 1. 获取当前楼层的消息内容
            const messageId = getCurrentMessageId();
            const messages = getChatMessages(messageId);
            const rawText = messages[0]?.message || '';

            // 如果消息体没有变化，则跳过本次更新以节省性能
            if (rawText === lastRawText) return;
            lastRawText = rawText;

            // ==========================================
            // 版块一：提取并渲染 <maintext>
            // ==========================================
            const mainTextMatch = rawText.match(/<maintext>([\s\S]*?)<\/maintext>/g);
            let mainText = '';
            if (mainTextMatch && mainTextMatch.length > 0) {
                // 取最后一个匹配项，剥离标签并去除首尾空白
                mainText = mainTextMatch[mainTextMatch.length - 1].replace(/<\/?maintext>/g, '').trim();
            } else {
                mainText = '（AI 仍在思考或未生成正文标签...）';
            }
            // 将换行符转换为 <br> 以便在 HTML 中正确渲染
            document.getElementById('mainTextContent').innerHTML = mainText.replace(/\n/g, '<br>');

            // ==========================================
            // 版块二：提取并渲染 <option>
            // ==========================================
            const optionMatch = rawText.match(/<option>([\s\S]*?)<\/option>/g);
            let options = [];
            if (optionMatch && optionMatch.length > 0) {
                const rawOptions = optionMatch[optionMatch.length - 1].replace(/<\/?option>/g, '');
                options = rawOptions.split('\n')
                                    .map(s => s.trim())
                                    .filter(s => s.length > 0);
            }

            const optionsCard = document.getElementById('optionsCard');
            const optionsList = document.getElementById('optionsList');
            
            // 清空旧按钮
            optionsList.innerHTML = '';
            
            if (options.length > 0) {
                optionsCard.style.display = 'block'; // 显示选项面板
                
                options.forEach((optText) => {
                    const btn = document.createElement('button');
                    btn.className = 'option-btn';
                    btn.innerHTML = `<i class="fa-solid fa-caret-right"></i> <span>${optText}</span>`;
                    
                    // 绑定点击事件，通过 triggerSlash 发送指令到酒馆输入框
                    btn.onclick = () => {
                        triggerSlash(`/setinput ${optText}`);
                    };
                    optionsList.appendChild(btn);
                });

                // 如果选项刷新了，默认展开以便用户查看
                if (!isOptionsExpanded) {
                    toggleOptions();
                }
            } else {
                optionsCard.style.display = 'none'; // 没有选项时隐藏面板
            }

            // ==========================================
            // 版块三：更新 MVU 状态栏
            // ==========================================
            const mvuData = Mvu.getMvuData({ type: 'message', message_id: messageId });
            const stat = _.get(mvuData, 'stat_data', {});

            const haogandu = Number(stat['好感度']) || 0;
            const haoganduBeizhu = stat['好感度备注'] || '暂无数据';
            const guanxi = stat['关系阶段'] || '未知';
            const xinqing = stat['心情'] || '未知';
            const weizhi = stat['当前位置'] || '未知';
            const chuangzhe = stat['穿着'] || '暂无数据';
            const shijian = stat['今日事件'] || '暂无记录';
            const neixin = stat['内心状态'] || '...';

            document.getElementById('var-haogandu-text').innerText = `${haogandu}/100`;
            document.getElementById('var-haogandu-bar').style.width = Math.min(Math.max(haogandu, 0), 100) + '%';
            document.getElementById('var-haogandu-beizhu').innerText = `(${haoganduBeizhu})`;

            document.getElementById('var-guanxi').innerText = guanxi;
            document.getElementById('var-xinqing').innerText = xinqing;
            document.getElementById('var-weizhi').innerText = weizhi;
            document.getElementById('var-chuangzhe').innerText = chuangzhe;
            document.getElementById('var-shijian').innerText = shijian;
            document.getElementById('var-neixin').innerText = neixin;

        } catch (error) {
            console.error('前端状态栏更新失败:', error);
        }
    }

    // 初始化入口
    $(async () => {
        // 等待全局 MVU 变量框架初始化完成
        await waitGlobalInitialized('Mvu');
        // 立即执行一次渲染
        updateDisplay();
        // 启动轮询，每 1 秒检查一次内容更新 (AI 生成过程中可实时渲染)
        setInterval(updateDisplay, 1000);
    });
</script>
