    <script>
        // 【隔离渲染规范】自执行匿名函数防止变量污染
        (function() {
            // 锁定当前面板的包裹层
            const currentScript = document.currentScript;
            const wrapper = currentScript.closest('.th-status-wrapper');

            // 注册翻转点击事件
            const innerCard = wrapper.querySelector('.flip-card-inner');
            wrapper.addEventListener('click', () => {
                innerCard.classList.toggle('is-flipped');
            });

            // ---------- 获取最新楼层的消息变量 ----------
            let data = null;
            try {
                // 检查全局函数是否存在（酒馆环境）
                if (typeof getVariables === 'function' && typeof getCurrentMessageId === 'function') {
                    // 获取最新楼层（-1 或 'latest'）
                    data = getVariables({ type: 'message', message_id: 'latest' });
                    console.log('[状态面板] 成功获取最新楼层变量:', data);
                } else {
                    console.warn('[状态面板] 未检测到酒馆API，使用模拟数据（仅用于预览）');
                    // 备选：若不在酒馆中，使用默认示例数据
                    data = getFallbackData();
                }
            } catch (e) {
                console.error('[状态面板] 获取变量失败，使用模拟数据', e);
                data = getFallbackData();
            }

            // 如果获取的数据为空或非对象，则使用模拟数据
            if (!data || typeof data !== 'object') {
                data = getFallbackData();
            }

            // ---------- 渲染面板 ----------
            renderPanel(wrapper, data);

            // ---------- 辅助函数：模拟数据（仅用于离线预览） ----------
            function getFallbackData() {
                return {
                    "姓名": "樱岛麻衣",
                    "日期": "2026/08/02",
                    "星期": "周日",
                    "时间": "14:02",
                    "暗示次数": 0,
                    "暗示等级": "无",
                    "城市": "藤泽市",
                    "地点": "樱岛麻衣家·客厅",
                    "穿着": {
                        "上装": "米色背心",
                        "内搭": "无",
                        "下装": "米色居家短裤",
                        "袜子": "无",
                        "鞋子": "居家拖鞋",
                        "配饰": "兔子发夹",
                        "内衣": "粉色文胸",
                        "内裤": "粉色内裤"
                    },
                    "开发度": {
                        "口": 1,
                        "胸": 0,
                        "阴蒂": 0,
                        "阴道": 0
                    }
                };
            }

            // ---------- 渲染函数 ----------
            function renderPanel(wrapper, data) {
                // --- 渲染正面 ---
                wrapper.querySelector('.val-name').textContent = data['姓名'] || '未命名';
                // 日期、星期、时间合并
                const dateTime = `${data['日期'] || ''} ${data['星期'] || ''} ${data['时间'] || ''}`.trim() || '未知时间';
                wrapper.querySelector('.val-datetime').textContent = dateTime;
                // 城市、地点合并
                const location = `${data['城市'] || ''} ✦ ${data['地点'] || ''}`.trim() || '未知地点';
                wrapper.querySelector('.val-location').textContent = location;

                // 穿着 (每行一项，排除内衣内裤)
                const clothesList = wrapper.querySelector('.clothes-list');
                clothesList.innerHTML = ''; // 清空
                const wear = data['穿着'] || {};
                for (const [key, value] of Object.entries(wear)) {
                    if (key !== '内衣' && key !== '内裤') {
                        const item = document.createElement('div');
                        item.className = 'clothes-item';
                        item.innerHTML = `<span>${key}</span> <span>${value}</span>`;
                        clothesList.appendChild(item);
                    }
                }

                // --- 渲染背面 ---
                wrapper.querySelector('.val-hint-level').textContent = data['暗示等级'] || '无';
                wrapper.querySelector('.val-hint-count').textContent = data['暗示次数'] || 0;
                
                // 暗示进度条联动 (根据次数计算宽度，这里设定100为满值)
                const maxHintCount = 100; 
                const hintProgress = Math.min(((data['暗示次数'] || 0) / maxHintCount) * 100, 100);
                wrapper.querySelector('.hint-bar-fill').style.width = `${hintProgress}%`;

                // 隐秘穿着
                const underwearList = wrapper.querySelector('.underwear-list');
                underwearList.innerHTML = '';
                ['内衣', '内裤'].forEach(key => {
                    if (wear[key]) {
                        const item = document.createElement('div');
                        item.className = 'underwear-item';
                        item.innerHTML = `<span>${key}</span> <span>${wear[key]}</span>`;
                        underwearList.appendChild(item);
                    }
                });

                // 开发度 (每行一项，0-5级分段进度条)
                const devList = wrapper.querySelector('.dev-list');
                devList.innerHTML = '';
                const dev = data['开发度'] || {};
                for (const [key, value] of Object.entries(dev)) {
                    const item = document.createElement('div');
                    item.className = 'dev-item';
                    
                    // 构建 5 个阶段的区块
                    let segmentsHtml = '';
                    const level = Math.min(Math.max(Number(value) || 0, 0), 5);
                    for (let i = 1; i <= 5; i++) {
                        segmentsHtml += `<div class="dev-segment ${i <= level ? 'filled' : ''}"></div>`;
                    }

                    item.innerHTML = `
                        <div class="dev-label"><span>${key}</span> <span>Lv.${level}</span></div>
                        <div class="dev-bar">${segmentsHtml}</div>
                    `;
                    devList.appendChild(item);
                }
            }
        })();
    </script>
